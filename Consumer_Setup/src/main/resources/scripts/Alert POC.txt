CREATE DATABASE ALERT;
USE ALERT;
CREATE TABLE template_mst (

    id BIGINT AUTO_INCREMENT PRIMARY KEY,

    template_id VARCHAR(50) NOT NULL,
    message_type VARCHAR(20) NOT NULL,
    alert_type VARCHAR(255) NOT NULL,

    template_variable JSON,

    version INT DEFAULT 1,
    is_active BOOLEAN DEFAULT TRUE,

    flexi_field1 VARCHAR(255),
    flexi_field2 VARCHAR(255),
    flexi_field3 VARCHAR(255),

    created_by VARCHAR(50),
    created_on TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    updated_by VARCHAR(50),
    updated_on TIMESTAMP NULL,

    -- Unique template_id
    CONSTRAINT uk_template_id UNIQUE (template_id),

    -- Unique combination of template_id + message_type + alert_type
    CONSTRAINT uk_template_msg_alert UNIQUE (template_id, message_type, alert_type)
    
);
commit;
--
DELIMITER $$

CREATE TRIGGER prevent_template_update
BEFORE UPDATE ON template_mst
FOR EACH ROW
BEGIN
    SIGNAL SQLSTATE '45000'
    SET MESSAGE_TEXT = 'UPDATE operation not allowed on template_mst';
END$$

DELIMITER ;

DELIMITER $$

CREATE TRIGGER prevent_template_delete
BEFORE DELETE ON template_mst
FOR EACH ROW
BEGIN
    SIGNAL SQLSTATE '45000'
    SET MESSAGE_TEXT = 'DELETE operation not allowed on template_mst';
END$$

DELIMITER ;

COMMIT;

------ AUDIT DATABASE

CREATE TABLE audit_mst (

    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    event_id VARCHAR(100) NOT NULL,
    message_type VARCHAR(50),
    headers JSON,
    payload JSON,
    status VARCHAR(50),
    created_on TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    created_by VARCHAR(50) DEFAULT 'SYSTEM'

);

DELIMITER $$

CREATE TRIGGER trg_audit_mst_no_delete
BEFORE DELETE ON audit_mst
FOR EACH ROW
BEGIN
    SIGNAL SQLSTATE '45000'
    SET MESSAGE_TEXT = 'DELETE operation is not allowed on audit_mst';
END$$

DELIMITER ;

COMMIT;


-- ROUTING CONFIG TABLE

CREATE TABLE KEY_ROUTING_CONFIG (
    
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    
    message_type VARCHAR(50) NOT NULL,
    
    alert_type VARCHAR(255) NOT NULL,
    
    event_type VARCHAR(255),
    
    application VARCHAR(255),
    
    sourceUri VARCHAR(255),
    
    version INT DEFAULT 1,
    
    is_active BOOLEAN DEFAULT TRUE,
    
    created_by VARCHAR(50) DEFAULT 'SYSTEM',
    
    created_on TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    
    updated_by VARCHAR(50) DEFAULT 'SYSTEM',
    
    updated_on TIMESTAMP DEFAULT CURRENT_TIMESTAMP
    
);

DELIMITER $$

CREATE TRIGGER key_routing_config_no_update
BEFORE UPDATE ON KEY_ROUTING_CONFIG
FOR EACH ROW
BEGIN
    SIGNAL SQLSTATE '45000'
    SET MESSAGE_TEXT = 'UPDATE operation is not allowed on KEY_ROUTING_CONFIG';
END$$

DELIMITER ;
